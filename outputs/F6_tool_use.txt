
----Defining a tool and sending a message that will make Claude ask for tool use----

{
    "role": "assistant",
    "content": [
        {
            "toolUse": {
                "toolUseId": "tooluse_nDnfWbiu6DmcpPYIGaNGbU",
                "name": "cosine",
                "input": {
                    "x": 7
                }
            }
        }
    ]
}

----Calling a function based on the toolUse content block.----

Using tool cosine
0.7539022543433046

----Passing the tool result back to Claude----

[
    {
        "role": "user",
        "content": [
            {
                "text": "What is the cosine of 7?"
            }
        ]
    },
    {
        "role": "assistant",
        "content": [
            {
                "toolUse": {
                    "toolUseId": "tooluse_nDnfWbiu6DmcpPYIGaNGbU",
                    "name": "cosine",
                    "input": {
                        "x": 7
                    }
                }
            }
        ]
    },
    {
        "role": "user",
        "content": [
            {
                "toolResult": {
                    "toolUseId": "tooluse_nDnfWbiu6DmcpPYIGaNGbU",
                    "content": [
                        {
                            "json": {
                                "result": 0.7539022543433046
                            }
                        }
                    ]
                }
            }
        ]
    },
    {
        "role": "assistant",
        "content": [
            {
                "text": "The cosine of 7 is approximately **0.754** (or more precisely, 0.7539022543433046).\n\nNote: This assumes the angle is in radians. If you meant 7 degrees, please let me know and I can calculate that instead."
            }
        ]
    }
]
